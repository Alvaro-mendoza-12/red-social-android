  const base = 'https://conectate.tail9b267c.ts.net';
  let account;

    const created = page.waitForResponse(response => response.url().endsWith('/v1/auth/register'));

    if (!account) {
      const recovered = await page.request.post(base + '/v1/auth/login', {

      });
      if(recovered.status() === 200) account = await recovered.json();

      const me = await page.request.get(base+'/v1/auth/me',{headers});
      const own = await me.json();
      if(own.user?.id !== account.user.id || own.user?.email !== email) throw new Error('Fixture ownership mismatch');
      const deleted = await page.request.post(base+'/rest/v1/rpc/delete_own_account',{headers,data:{}});
      if(deleted.status() !== 200) throw new Error('Fixture cleanup failed');
    }
    await page.goto('https://red-social-android.vercel.app/login');
  }
}
